    import numpy as np

# Scratch implementation
class Vector:
    def __init__(self, components):
        self.data = components
        self.dim = len(components)
    def __add__(self, other):
        return Vector([a + b for a, b in zip(self.data, other.data)])
    def magnitude(self):
        return sum(x**2 for x in self.data) ** 0.5
    def dot(self, other):
        return sum(a * b for a, b in zip(self.data, other.data))

# NumPy (professional way)
a, b = np.array([1, 2]), np.array([3, 1])
print(a + b)               # [4, 3]
print(np.linalg.norm(a))   # 2.2361
print(np.dot(a, b))        # 5

# Linear independence check
def are_linearly_independent(vectors):
    matrix = np.stack(vectors, axis=0)
    return np.linalg.matrix_rank(matrix) == len(vectors)

print(are_linearly_independent([np.array([1,0]), np.array([0,1])]))  # True
print(are_linearly_independent([np.array([1,0]), np.array([2,0])]))  # False
